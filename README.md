  - backend

    - Spring security
    - MapStruct
    - robsonkades.UUIDv7

  - backend-testing

    - RestAssured
  - frontend
    - Angular
    - Swagger CLI generator

    - Ansible
    - Docker

- External services
  - PostgreSQL
  - Redis

- frontend
